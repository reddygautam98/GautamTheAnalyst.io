SQL SCRIPT sample_reconciliation_queries.sql 
-- Identify unmatched records breaching 4-hour triage SLA
SELECT pt.payment_id, pt.amount, ge.amount AS gateway_amount,
       ROUND(CAST(EXTRACT(EPOCH FROM (CURRENT_TIMESTAMP - ec.created_at)) / 3600.0 AS numeric), 1) AS hours_open,
       CASE
         WHEN ge.event_id IS NULL THEN 'MISSING_GATEWAY_EVENT'
         WHEN pt.amount <> ge.amount THEN 'AMOUNT_MISMATCH'
         ELSE 'OTHER'
       END AS mismatch_reason
FROM PAYMENT_TRANSACTION pt
LEFT JOIN GATEWAY_EVENT ge ON pt.payment_id = ge.payment_id
JOIN EXCEPTION_CASE ec ON pt.payment_id = ec.payment_id
WHERE pt.status = 'UNMATCHED' AND ec.case_status <> 'RESOLVED'
ORDER BY hours_open DESC LIMIT 4;
