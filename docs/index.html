SQL SCRIPTsample_reconciliation_queries.sql
-- Surface un-reconciled records approaching 4h SLA breach
SELECT t.txn_id, t.amount, t.currency,
       ROUND(EXTRACT(EPOCH FROM (NOW() - t.created_at))/3600, 1) AS pending_hrs,
       CASE
         WHEN b.bank_ref IS NULL THEN 'MISSING_BANK_FEED'
         WHEN b.amount != t.amount THEN 'AMOUNT_MISMATCH'
       END AS discrepancy_reason
FROM core_transactions t
LEFT JOIN bank_settlement_feed b ON t.gateway_ref = b.gateway_ref
WHERE t.status = 'PENDING_SETTLEMENT'
  AND (NOW() - t.created_at) > INTERVAL '3 hours'
ORDER BY pending_hrs DESC LIMIT 4;
